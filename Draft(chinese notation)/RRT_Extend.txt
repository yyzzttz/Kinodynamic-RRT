FUNCTION RRT_SINGLE_EXTEND(q_start, q_goal,step_size, goal_eps, max_iters):

    # 1. 初始化树 T，根节点为 q_start
    T ← empty tree
    root ← NewNode(q_start)
    root.parent ← NULL
    AddNode(T, root)

    # 2. 外层主循环：最多迭代 max_iters 次，这个循环做的是每轮随机采样一个q_rand让已有树T去接近它
    FOR k FROM 1 TO max_iters DO

        # 2.1 随机采样一个配置 q_rand
        q_rand ← SampleConfig()

        # 2.2 用 EXTEND_BASIC 向 q_rand 方向扩展一步。
        #这一步返回值是（1）failure，对应在extend_basic中监测到有碰撞，没有走出去这一步，那么对应的在RRTextend里这一轮就失败了，跳到下一轮去，重新选q_rand，再做extend
        #这一步返回值是（2）new_node，对应在extend_basic中没碰撞，成功步进，并扩展了树。那么就进入接下来判断这次步进后有没有足够接近q_rand
        new_node ← EXTEND_BASIC(T, q_rand, step_size)

        # 对应返回值（1）。若这一步扩展失败（立刻碰障），本轮什么也不做，进入下一轮迭代
        IF new_node == FAILURE THEN
            CONTINUE

        # 对应返回值（2）。现在来判断这次extend有没有足够接近q_goal
        # 2.3 如果新节点已经足够接近目标 q_goal，就结束
        IF Distance(new_node.q, q_goal) < goal_eps THEN

            # 从 new_node 沿 parent 指针回溯完整路径
            path ← BacktracePath(new_node)   # 得到 q_start → new_node 的序列
            RETURN path                      # 算法整体结束

    END FOR

    # 超过 max_iters 仍然找不到接近 q_goal 的节点
    RETURN FAILURE
END FUNCTION

这里的extend方法就是完整地调用了extend_basic循环，每次循环迭代采样q_rand作为目标点去extend，并判断未碰撞的new_node是否足够接近q_goal
退出条件：当某一轮发现new_node足够接近q_goal的时候，成功并推出；跑完max_iter还没有形成连通树，失败并退出
