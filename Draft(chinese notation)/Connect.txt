FUNCTION CONNECT(T, q_target, step_size, goal_eps):
    # T: 当前树（单棵）
    # q_target: 固定目标（可以是 q_goal，也可以是某个点）
    # step_size: 单步步长
    # goal_eps: 判断“到达 q_target”的距离阈值
    # 返回：(status, last_node)
    #   status ∈ {"REACHED", "TRAPPED"}
    #   last_node: 最后一次成功加入树的节点（可能为 NULL）

    last_node ← NULL

    WHILE TRUE DO
        # 每一轮：在当前树 T 上，向 q_target 方向尝试走一步
        # 时刻牢记这个函数所返回的东西是failure或者步进一步之后的new_node
        new_node ← EXTEND_BASIC(T, q_target, step_size)

        # ---- 退出条件 1：TRAPPED ----
        IF new_node == FAILURE THEN
            # 表示再往 q_target 方向多走一点就会碰障 → 走不动了
            RETURN ("TRAPPED", last_node)

        # 走出了一小步，记录最新节点
        last_node ← new_node

        # ---- 退出条件 2：REACHED ----
        IF Distance(new_node.q, q_target) < goal_eps THEN
            # 已经足够接近 q_target → 认为到达
            RETURN ("REACHED", last_node)

        # ---- 否则：继续 while ----
        #意思是：
        # 还能走，而且离目标还远 → 下一轮 while 再次调用 EXTEND_BASIC
        # 在新的树结构基础上继续走一步
    END WHILE
END FUNCTION

connect是在extend_basic只步进一步的基础上演变成朝q_rand尽可能多的走，直到走不动为止。
对于被采中的q_rand，每一次步进要么碰障碍trapped（对应extend_basic里的failure），要么newnode离qgaol够近reached，要么就是还没到还能走advanced（此种情况继续执行循环）
advanced只作为一种中间状态控制持续步进，最终也只会return trapped或者reached两种结果
