FUNCTION RRT_SINGLE_CONNECT(q_start, q_goal,
                            step_size, goal_eps, max_iters):

    # 1. 初始化单棵树 T，从起点出发
    T ← empty tree
    root ← NewNode(q_start)
    root.parent ← NULL
    AddNode(T, root)

    # 2. 外层主循环
    FOR k FROM 1 TO max_iters DO

        # 2.1 像普通 RRT 一样，随机探索一下空间
        q_rand ← SampleConfig()
        new_node ← EXTEND_BASIC(T, q_rand, step_size)
        # 这一步和普通单树 RRT 一样，可能失败也可能成功

        # 如果这一步失败，就直接进行下一轮随机采样
        IF new_node == FAILURE THEN
            CONTINUE

        # 2.2 在“已经扩展过的树”基础上，尝试用 CONNECT 朝 q_goal 靠近
        (status, last_node) ← CONNECT(T, q_goal,
                                      step_size, goal_eps)

        # 如果 CONNECT 成功一路冲到 q_goal 附近，就结束
        IF status == "REACHED" THEN

            # 通过 last_node 回溯出从 q_start 到 q_goal 的路径
            path ← BacktracePath(last_node)
            RETURN path      # 算法结束

        # 否则（TRAPPED），说明在当前树结构下，
        # 再往 q_goal 方向已经撞障走不动 → 下一轮 FOR 再随机探索别的地方
        # 进入下一个 k

    END FOR

    # 超过 max_iters 还没连接到 q_goal
    RETURN FAILURE
END FUNCTION

这里是完整的单树条件下把connect放到每个采样迭代里形成完整的过程。
每个迭代里connect返回trapped时说明这次connect里某个extend存在遇到障碍，这一次connect都会退出并进入下一个for进入下一次随机探索（continue效果），如果reached了就返回路径
